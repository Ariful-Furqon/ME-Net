\subsection{Notation and Problem Statement}
\label{sec:notation}

Let $\mathcal{V}=\{1,\dots,V\}$ be a token vocabulary with a reserved padding index $0$, and let a document
be a token sequence $\mathbf{x}=(x_1,\dots,x_L)\in\{0\}\cup\mathcal{V}^{L}$ truncated or zero-padded to a fixed
length $L$. We write $w(\mathbf{x})=(w_1,\dots,w_N)$ for the untruncated word sequence of the same document
(used by the symbolic stream, so that the profile is not affected by truncation), $N=|w(\mathbf{x})|$, and
$\mathcal{L}\subset\Sigma^{*}$ for a lexicon of Indonesian root words. Given a labelled corpus
$\mathcal{D}=\{(\mathbf{x}^{(i)},y^{(i)})\}_{i=1}^{n}$ with $y\in\{0,1\}$ ($1=$ hoax), we learn
\begin{equation}
f_{\bm{\theta}}:\{0\}\cup\mathcal{V}^{L}\longrightarrow[0,1],\qquad
f_{\bm{\theta}}(\mathbf{x})=\hat{p}(y=1\mid\mathbf{x}),
\end{equation}
as the composition
\begin{equation}
f_{\bm{\theta}}
=\underbrace{c_{\bm{\theta}_c}}_{\text{classifier}}\circ
\underbrace{\mathcal{F}_{\bm{\theta}_f}}_{\text{gated fusion}}\circ
\Big(\underbrace{e_{\bm{\theta}_t}}_{\text{contextual}}\;,\;
\underbrace{m_{\bm{\theta}_\ell}\circ\mathcal{S}\circ\bm{\Phi}}_{\text{symbolic}}\Big),
\end{equation}
where $\bm{\Phi}$ is a \emph{fixed, parameter-free} linguistic feature map, $\mathcal{S}$ is standardisation,
and $\bm{\theta}=(\bm{\theta}_t,\bm{\theta}_\ell,\bm{\theta}_f,\bm{\theta}_c)$. Symbols are summarised in
Table~\ref{tab:notation}.

\subsection{The Morpho-Evidential Feature Map \texorpdfstring{$\bm{\Phi}$}{Phi}}
\label{sec:featuremap}

Indonesian marks voice, transitivity and nominalisation with productive affixes
\citep{sneddon2010indonesian} and marks information source lexically rather than grammatically
\citep{aikhenvald2004evidentiality}. $\bm{\Phi}:\Sigma^{*}\to\mathbb{R}^{m}$, $m=13$, turns a document into
four groups of interpretable statistics.

\paragraph{Counting operator.}
For a predicate $\pi:\Sigma^{*}\to\{0,1\}$ over word tokens, write the count and the length-normalised
rate (occurrences per 100 words) as
\begin{equation}
c_\pi(\mathbf{x})=\sum_{i=1}^{N}\pi(w_i),
\qquad
r_\pi(\mathbf{x})=\frac{100}{\max(N,1)}\,c_\pi(\mathbf{x}).
\label{eq:rate}
\end{equation}

\paragraph{Lexicon-verified affix predicates.}
Let $\mathrm{Pre}_{\text{pas}}=\{\textit{di},\textit{ter},\textit{te}\}$ and
$\mathrm{Pre}_{\text{act}}=\{\textit{meng},\textit{meny},\textit{mem},\textit{men},\textit{me},\textit{ber},\textit{bel},\textit{be}\}$
be ordered prefix sets, and let $\mathrm{All}(p)\subset\Sigma^{*}$ be the set of consonants restored when
prefix $p$ triggers nasal assimilation, e.g.\
$\mathrm{All}(\textit{meny})=\{s\}$, $\mathrm{All}(\textit{meng})=\{\varepsilon,k\}$,
$\mathrm{All}(\textit{mem})=\{\varepsilon,p\}$, $\mathrm{All}(\textit{men})=\{\varepsilon,t\}$, and
$\mathrm{All}(p)=\{\varepsilon\}$ otherwise. For a token $t$ and an exception set $\mathcal{E}_\bullet$,
\begin{equation}
\pi_{\text{pas}}(t)=\mathbb{1}\Big[|t|>3\;\wedge\;t\notin\mathcal{E}_{\text{pas}}\;\wedge\;
\exists\,p\in\mathrm{Pre}_{\text{pas}},\,\exists\,c\in\mathrm{All}(p):\;
t=p\!\cdot\!\rho\;\wedge\; c\!\cdot\!\rho\in\mathcal{L}\Big],
\label{eq:passive}
\end{equation}
where $\cdot$ denotes string concatenation and $\rho$ the residue after stripping $p$;
$\pi_{\text{act}}$ is defined identically with $\mathrm{Pre}_{\text{act}}$ and
$\mathcal{E}_{\text{act}}$. The circumfix predicate additionally requires both edges,
\begin{equation}
\pi_{\text{ke-an}}(t)=\mathbb{1}\Big[t=\textit{ke}\!\cdot\!\rho\!\cdot\!\textit{an}\;\wedge\;|t|>5\;\wedge\;
t\notin\mathcal{E}_{\text{ke-an}}\;\wedge\;\big(\rho\in\mathcal{L}\vee t\in\mathcal{K}\big)\Big],
\end{equation}
with $\mathcal{K}$ a list of nominalisations whose base is itself derived (e.g.\ \emph{keberhasilan}).
The exception sets contain every lexicon root that begins with a candidate prefix, which blocks the
dominant error mode of naïve prefix stripping: \emph{dinas} `office' is a root, not a passive of
\emph{nas}; \emph{menteri} `minister' is a root, not an active form of \emph{teri} `anchovy'. Membership
tests are $O(1)$ hash lookups, so \eqref{eq:passive} needs no full stemmer
\citep{adriani2007stemming}; per-token decisions are memoised, and each token is assigned to at most one
of the three morphological categories (passive $\succ$ active $\succ$ circumfix).

\paragraph{Discourse particles.}
With free particles $\mathcal{P}$ (\emph{kok, sih, dong, deh, lho, toh, \ldots}), bound clitics
$\mathcal{C}=\{\textit{-lah},\textit{-kah},\textit{-tah},\textit{-pun}\}$ and lexicalised look-alikes
$\mathcal{E}_{\text{cl}}$ (\emph{masalah}, \emph{sekolah}, \emph{walaupun}),
\begin{equation}
\pi_{\text{prt}}(t)=\mathbb{1}\big[t\in\mathcal{P}\big]\;\vee\;
\mathbb{1}\Big[t\notin\mathcal{E}_{\text{cl}}\cup\mathcal{L}\;\wedge\;
\exists\,s\in\mathcal{C}:\,t=\beta\!\cdot\!s,\;|\beta|>1,\;\beta\in\mathcal{L}\Big].
\end{equation}
Possessive \emph{-nya} is excluded, since it marks definiteness rather than stance.

\paragraph{Evidential and register markers.}
Three disjoint regular-expression lexicons $\mathcal{R}_{\text{rum}}$ (17 rumour triggers:
\emph{dikabarkan, konon, katanya, bocoran, viralkan}, \ldots), $\mathcal{R}_{\text{ins}}$ (14 attribution
triggers: \emph{menurut, ujar, juru bicara, siaran pers}, \ldots) and $\mathcal{R}_{\text{sen}}$
(17 sensational markers: \emph{heboh, gawat, merinding}, \ldots) are matched against the lower-cased
document with word boundaries, giving rates $r_{\text{rum}},r_{\text{ins}},r_{\text{sen}}$ by
\eqref{eq:rate}. Their contrast is summarised by a smoothed ratio
\begin{equation}
\Psi(\mathbf{x})=\frac{r_{\text{ins}}(\mathbf{x})+\epsilon}{r_{\text{rum}}(\mathbf{x})+\epsilon},
\qquad \epsilon=10^{-2},
\label{eq:ratio}
\end{equation}
which is finite for every document, equals $1$ when both rates vanish, and grows with the dominance of
institutional sourcing over rumour framing.

\paragraph{Assembled profile.}
\begin{equation}
\bm{\Phi}(\mathbf{x})=\big[\underbrace{N,\;|\mathbf{x}|_{\text{ch}},\;\bar{\lambda},\;\mathrm{TTR},\;\mathrm{HPX}}_{\text{surface}},\;
\underbrace{r_{\text{pas}},\,r_{\text{act}},\,r_{\text{ke-an}},\,r_{\text{prt}}}_{\text{morphological}},\;
\underbrace{r_{\text{rum}},\,r_{\text{ins}},\,r_{\text{sen}},\,\Psi}_{\text{evidential}}\big]^{\top}\in\mathbb{R}^{13},
\end{equation}
where $\bar{\lambda}=\frac{1}{N}\sum_i|w_i|$ is mean word length,
$\mathrm{TTR}=|\{w_i\}|/N$ the type--token ratio, and
$\mathrm{HPX}=|\{v:\,c_v=1\}|/N$ the hapax ratio.

\begin{proposition}[Scale behaviour]
\label{prop:scale}
Let $\mathbf{x}^{\oplus 2}$ be the concatenation of a document with itself. Then
$r_\pi(\mathbf{x}^{\oplus 2})=r_\pi(\mathbf{x})$ for every predicate rate and $\Psi(\mathbf{x}^{\oplus2})=\Psi(\mathbf{x})$,
whereas $N$ and $|\mathbf{x}|_{\text{ch}}$ double, $\mathrm{TTR}$ halves and $\mathrm{HPX}$ drops to $0$.
The same holds for $\bar\lambda$, which is invariant.
\end{proposition}
\begin{proof}
Counts are additive over concatenation, $c_\pi(\mathbf{x}^{\oplus2})=2c_\pi(\mathbf{x})$, and $N$ doubles, so the
factor $100/N$ cancels in \eqref{eq:rate} and hence in \eqref{eq:ratio}; the same cancellation gives
$\bar\lambda$. The type set is unchanged while $N$ doubles, so $\mathrm{TTR}$ halves, and every type now
occurs an even number of times, so no hapax remains.
\end{proof}

Proposition~\ref{prop:scale} makes explicit which coordinates of $\bm{\Phi}$ are length-invariant and which
are not: the two raw counts are the only features that encode document length directly. Because corpus
construction can correlate length with the label, we treat them as a separate, ablatable group
(\S\ref{sec:setup}).

\paragraph{Standardisation.}
With training-split statistics $\bm{\mu},\bm{\sigma}\in\mathbb{R}^{m}$,
\begin{equation}
\mathcal{S}(\bm{\Phi})=\big(\bm{\Phi}-\bm{\mu}\big)\oslash\bm{\sigma},
\end{equation}
where $\oslash$ is element-wise division. $\bm{\mu},\bm{\sigma}$ are estimated on training data only.

\subsection{Contextual Stream}
\label{sec:contextual}

Tokens are embedded by $\mathbf{E}\in\mathbb{R}^{V\times d_e}$ (row $0$ fixed to $\mathbf{0}$), giving
$\mathbf{Z}=[\mathbf{E}_{x_1},\dots,\mathbf{E}_{x_L}]^{\top}\in\mathbb{R}^{L\times d_e}$. A width-$k$ convolution with
`same' padding produces, for filter $j\in\{1,\dots,F\}$ and position $i$,
\begin{equation}
a_{ij}=\sum_{u=1}^{k}\sum_{v=1}^{d_e}W^{(c)}_{j,v,u}\,Z_{\,i+u-\lceil k/2\rceil,\;v}+b^{(c)}_j ,
\label{eq:conv}
\end{equation}
followed by batch normalisation \citep{ioffe2015bn}, a ReLU and global max-pooling over positions:
\begin{equation}
\mathbf{h}_t=\operatorname{Drop}_{p}\Big(\big[\max_{1\le i\le L}\operatorname{ReLU}\big(\gamma_j\hat{a}_{ij}+\beta_j\big)\big]_{j=1}^{F}\Big)\in\mathbb{R}_{\ge0}^{F},
\label{eq:hcontext}
\end{equation}
with $\hat{a}_{ij}$ the batch-normalised pre-activation. Each coordinate of $\mathbf{h}_t$ is the strongest
response of one learned $k$-gram detector anywhere in the document, a single-width instance of the
sentence CNN of \citet{kim2014cnn}; max-pooling makes $\mathbf{h}_t$ invariant to where a cue occurs but
sensitive to whether it occurs at all.

\subsection{Symbolic Stream}
\label{sec:symbolic}

The standardised profile is encoded by a two-layer perceptron
\begin{equation}
\mathbf{h}_\ell=\operatorname{ReLU}\Big(\mathbf{W}^{(2)}\operatorname{ReLU}\big(\operatorname{BN}(\mathbf{W}^{(1)}\mathcal{S}(\bm{\Phi}(\mathbf{x}))+\mathbf{b}^{(1)})\big)+\mathbf{b}^{(2)}\Big)\in\mathbb{R}_{\ge0}^{d_\ell},
\end{equation}
with $\mathbf{W}^{(1)}\in\mathbb{R}^{d_\ell\times m}$ and $\mathbf{W}^{(2)}\in\mathbb{R}^{d_\ell\times d_\ell}$. The map
is low-dimensional by design: it can learn interactions among the 13 markers (for instance, high rumour
rate \emph{and} low attribution rate) without the capacity to memorise lexical content.

\subsection{Gated Cross-Stream Fusion}
\label{sec:fusion}

Both streams are first projected into a shared $d$-dimensional space,
\begin{equation}
\mathbf{z}_t=\operatorname{ReLU}(\mathbf{P}_t\mathbf{h}_t+\mathbf{b}_t)\in\mathbb{R}_{\ge0}^{d},\qquad
\mathbf{z}_\ell=\operatorname{ReLU}(\mathbf{P}_\ell\mathbf{h}_\ell+\mathbf{b}_\ell)\in\mathbb{R}_{\ge0}^{d},
\end{equation}
and a gate conditioned on \emph{both} views assigns each of the $d$ coordinates a mixing weight
\begin{equation}
\boxed{\;\mathbf{g}(\mathbf{x})=\sigma\big(\mathbf{W}_g[\mathbf{z}_t;\mathbf{z}_\ell]+\mathbf{b}_g\big)\in(0,1)^{d},\qquad
\mathbf{W}_g\in\mathbb{R}^{d\times 2d}.\;}
\label{eq:gate}
\end{equation}
The gate modulates the two projections \emph{complementarily} and the result is compressed back to $d$:
\begin{align}
\mathbf{u}&=\big[\mathbf{g}\odot\mathbf{z}_t\;;\;(\mathbf{1}-\mathbf{g})\odot\mathbf{z}_\ell\big]\in\mathbb{R}^{2d},\label{eq:u}\\
\mathbf{h}&=\operatorname{Drop}_{p_f}\Big(\operatorname{ReLU}\big(\operatorname{LN}(\mathbf{W}_u\mathbf{u}+\mathbf{b}_u)\big)\Big)\in\mathbb{R}^{d}.\label{eq:fuse}
\end{align}
Writing $\mathbf{W}_u=[\mathbf{A}\;\;\mathbf{B}]$ with $\mathbf{A},\mathbf{B}\in\mathbb{R}^{d\times d}$, the pre-norm fused
state is
\begin{equation}
\mathbf{W}_u\mathbf{u}=\mathbf{A}\big(\mathbf{g}\odot\mathbf{z}_t\big)+\mathbf{B}\big((\mathbf{1}-\mathbf{g})\odot\mathbf{z}_\ell\big),
\label{eq:bilinear}
\end{equation}
i.e.\ a \emph{multiplicative} interaction between a data-dependent gate and each stream, unlike the purely
additive interaction of concatenation. Three properties follow.

\begin{proposition}[Concatenation is a special case]
\label{prop:concat}
Let $\mathcal{H}_{\mathrm{cat}}$ be the family of maps $(\mathbf{z}_t,\mathbf{z}_\ell)\mapsto
\mathbf{A}'\mathbf{z}_t+\mathbf{B}'\mathbf{z}_\ell+\mathbf{b}$ realised by naive concatenation followed by a linear layer,
and $\mathcal{H}_{\mathrm{gate}}$ the family realised by \eqref{eq:gate}--\eqref{eq:bilinear}. Then
$\mathcal{H}_{\mathrm{cat}}\subseteq\mathcal{H}_{\mathrm{gate}}$.
\end{proposition}
\begin{proof}
Set $\mathbf{W}_g=\mathbf{0}$, so that $\mathbf{g}=\sigma(\mathbf{b}_g)=:\bm{\alpha}\in(0,1)^d$ is constant in
$\mathbf{x}$. Then \eqref{eq:bilinear} becomes
$\mathbf{A}\operatorname{diag}(\bm{\alpha})\mathbf{z}_t+\mathbf{B}\operatorname{diag}(\mathbf{1}-\bm{\alpha})\mathbf{z}_\ell$.
Since $\alpha_j\in(0,1)$ strictly, $\operatorname{diag}(\bm{\alpha})$ and
$\operatorname{diag}(\mathbf{1}-\bm{\alpha})$ are invertible, so choosing
$\mathbf{A}=\mathbf{A}'\operatorname{diag}(\bm{\alpha})^{-1}$ and
$\mathbf{B}=\mathbf{B}'\operatorname{diag}(\mathbf{1}-\bm{\alpha})^{-1}$ reproduces any
$(\mathbf{A}',\mathbf{B}')$.
\end{proof}

Proposition~\ref{prop:concat} says the gate can only add expressive power; whether that power helps is an
empirical question, which is exactly what the \emph{w/o gate} ablation in \S\ref{sec:results} measures.
The gated multimodal unit of \citet{arevalo2017gated} is recovered from \eqref{eq:bilinear} by the
further constraint $\mathbf{A}=\mathbf{B}=\mathbf{I}$; keeping $\mathbf{A},\mathbf{B}$ free lets the two streams
be read out along different directions before they are summed.

\begin{proposition}[Gated gradient routing]
\label{prop:grad}
For the pre-norm state $\mathbf{s}=\mathbf{W}_u\mathbf{u}+\mathbf{b}_u$ and a loss $\mathcal{L}$,
\begin{equation}
\frac{\partial \mathbf{s}}{\partial \mathbf{z}_t}=\mathbf{A}\operatorname{diag}(\mathbf{g})+\mathbf{R}_t,
\qquad
\frac{\partial \mathbf{s}}{\partial \mathbf{z}_\ell}=\mathbf{B}\operatorname{diag}(\mathbf{1}-\mathbf{g})+\mathbf{R}_\ell,
\end{equation}
with the gate-feedback terms
$\mathbf{R}_t=\big(\mathbf{A}\operatorname{diag}(\mathbf{z}_t)-\mathbf{B}\operatorname{diag}(\mathbf{z}_\ell)\big)\operatorname{diag}\big(\mathbf{g}\odot(\mathbf{1}-\mathbf{g})\big)\mathbf{W}_g^{[t]}$
and $\mathbf{R}_\ell$ defined analogously with $\mathbf{W}_g^{[\ell]}$, where $\mathbf{W}_g=[\mathbf{W}_g^{[t]}\;\mathbf{W}_g^{[\ell]}]$.
\end{proposition}
\begin{proof}
Differentiate \eqref{eq:bilinear} using $\partial\sigma(\cdot)/\partial\cdot=\mathbf{g}\odot(\mathbf{1}-\mathbf{g})$
and the product rule; the first term treats $\mathbf{g}$ as constant, the second collects its dependence on
$\mathbf{z}_t$ and $\mathbf{z}_\ell$.
\end{proof}

Thus a coordinate with $g_j\!\to\!0$ receives essentially no gradient through the contextual path and the
symbolic path is trained at full strength, and vice versa: the gate performs per-document, per-dimension
credit assignment between the two views. Because the modulation is complementary
($g_j+(1-g_j)=1$), no coordinate can amplify both streams at once: since $\mathbf{g}\in(0,1)^d$ acts
element-wise, $\lVert\mathbf{u}\rVert_2\le(\lVert\mathbf{z}_t\rVert_2^2+\lVert\mathbf{z}_\ell\rVert_2^2)^{1/2}$ and
hence $\lVert\mathbf{W}_u\mathbf{u}\rVert_2\le\lVert\mathbf{W}_u\rVert_2(\lVert\mathbf{z}_t\rVert_2^2+\lVert\mathbf{z}_\ell\rVert_2^2)^{1/2}$,
so fusion cannot inflate the representation norm relative to its inputs.

\paragraph{An interpretable statistic.}
Averaging the gate gives a scalar \emph{text-reliance} score per document,
\begin{equation}
\bar{g}(\mathbf{x})=\frac{1}{d}\sum_{j=1}^{d}g_j(\mathbf{x})\in(0,1),
\label{eq:gbar}
\end{equation}
with $\bar g\to1$ meaning the prediction is carried by lexical evidence and $\bar g\to0$ meaning it is
carried by the morpho-evidential profile. Unlike post-hoc attribution, $\bar g$ is read directly off the
forward pass; we analyse its class-conditional distribution in \S\ref{sec:results}.

\subsection{Classifier and Objective}
\label{sec:objective}

The head is a two-layer perceptron producing a logit and a probability,
\begin{equation}
s=\mathbf{w}_2^{\top}\operatorname{Drop}_{p}\big(\operatorname{ReLU}(\mathbf{W}_3\mathbf{h}+\mathbf{b}_3)\big)+b_2,
\qquad \hat{p}=\sigma(s),
\end{equation}
trained end-to-end by minimising the regularised binary cross-entropy
\begin{equation}
\mathcal{L}(\bm{\theta})=-\frac{1}{|B|}\sum_{i\in B}\Big[y^{(i)}\log\hat p^{(i)}+(1-y^{(i)})\log(1-\hat p^{(i)})\Big]
\;+\;\frac{\lambda}{2}\lVert\bm{\theta}\rVert_2^{2},
\end{equation}
where the $\ell_2$ term is applied through decoupled weight decay \citep{loshchilov2019adamw}, i.e.\ the
AdamW update $\bm{\theta}\leftarrow\bm{\theta}-\eta\big(\hat{\mathbf{m}}/(\sqrt{\hat{\mathbf{v}}}+\varepsilon)+\lambda\bm{\theta}\big)$.
All gradients flow through $\mathbf{g}$; no auxiliary loss, gate supervision or annealing is used, so any
gate behaviour we observe is induced purely by the classification objective.

\begin{algorithm}[t]
\caption{\menet{} forward pass for one document}
\label{alg:forward}
\begin{algorithmic}[1]
\Require token ids $\mathbf{x}\in\{0,\dots,V\}^{L}$; raw text; training statistics $\bm{\mu},\bm{\sigma}$
\State $\bm{\phi}\gets\bm{\Phi}(\text{text})$ \Comment{single memoised pass over word tokens, $O(N)$}
\State $\tilde{\bm{\phi}}\gets(\bm{\phi}-\bm{\mu})\oslash\bm{\sigma}$
\State $\mathbf{h}_t\gets\textsc{MaxPool}\big(\operatorname{ReLU}(\operatorname{BN}(\operatorname{Conv}_k(\mathbf{E}_{\mathbf{x}})))\big)$ \Comment{Eqs.~\eqref{eq:conv}--\eqref{eq:hcontext}}
\State $\mathbf{h}_\ell\gets\textsc{MLP}_\ell(\tilde{\bm{\phi}})$
\State $\mathbf{z}_t\gets\operatorname{ReLU}(\mathbf{P}_t\mathbf{h}_t+\mathbf{b}_t)$;\quad $\mathbf{z}_\ell\gets\operatorname{ReLU}(\mathbf{P}_\ell\mathbf{h}_\ell+\mathbf{b}_\ell)$
\State $\mathbf{g}\gets\sigma(\mathbf{W}_g[\mathbf{z}_t;\mathbf{z}_\ell]+\mathbf{b}_g)$ \Comment{Eq.~\eqref{eq:gate}}
\State $\mathbf{h}\gets\operatorname{ReLU}\big(\operatorname{LN}(\mathbf{W}_u[\mathbf{g}\odot\mathbf{z}_t;(\mathbf{1}-\mathbf{g})\odot\mathbf{z}_\ell]+\mathbf{b}_u)\big)$
\State \Return $\hat p=\sigma(\mathbf{w}_2^{\top}\operatorname{ReLU}(\mathbf{W}_3\mathbf{h}+\mathbf{b}_3)+b_2)$, $\;\bar g=\tfrac1d\sum_j g_j$
\end{algorithmic}
\end{algorithm}

\subsection{Complexity}
\label{sec:complexity}

The parameter count decomposes as
\begin{equation}
|\bm{\theta}|=\underbrace{Vd_e}_{\text{embedding}}+\underbrace{kd_eF+F+2F}_{\text{conv}+\text{BN}}
+\underbrace{(m+1)d_\ell+2d_\ell+(d_\ell+1)d_\ell}_{\text{symbolic}}
+\underbrace{(F{+}1)d+(d_\ell{+}1)d+(2d{+}1)d+(2d{+}1)d+2d}_{\text{fusion}}
+\underbrace{(d{+}1)\tfrac d2+\tfrac d2+1}_{\text{head}},
\end{equation}
which for the defaults gives 4.03M parameters, of which 3.84M (95.4\%) is the embedding table
(Table~\ref{tab:params}). The symbolic stream and the entire gating mechanism cost 0.10M parameters, a
2.4\% increase over a text-only CNN with the same head. Time is dominated by the convolution,
$\Theta(Lkd_eF)$ multiply-accumulates per document ($\approx$21M for the defaults), versus
$\Theta(md_\ell+d_\ell^2)$ for the symbolic stream ($\approx$5K) and $\Theta(d^2)$ for fusion
($\approx$90K); feature extraction is $\Theta(N)$ with $O(1)$ amortised work per token. The symbolic
pathway is therefore effectively free at inference time.
